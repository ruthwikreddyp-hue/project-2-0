function render(){
  const d=new Date();
  dayPickerSync();
  document.getElementById('date').textContent=formatSelectedDate(selectedDay());
  const isToday=selectedDay()===today();
  const h=d.getHours();
  document.getElementById('greeting').textContent=isToday?(h<12?'Build the next version.':h<18?'Keep the day moving.':'Finish strong.'):'Viewing '+formatSelectedDate(selectedDay());
  document.getElementById('homeHeading').textContent=isToday?'Today':formatSelectedDate(selectedDay());

  const weightForDay=data.weights.filter(x=>x.date===selectedDay()).at(-1);
  document.getElementById('homeWeight').textContent=weightForDay?Number(weightForDay.value).toFixed(1)+' kg':'—';
